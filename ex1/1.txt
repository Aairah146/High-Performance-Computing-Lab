#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int **create(int r,int c)
{
    int **m=(int**)malloc(r*sizeof(int*));
    for(int i=0;i<r;i++)
        m[i]=(int*)malloc(c*sizeof(int));
    return m;
}

void manual(int **A,int r,int c)
{
    for(int i=0;i<r;i++)
        for(int j=0;j<c;j++)
            scanf("%d",&A[i][j]);
}

void randomFill(int **A,int r,int c)
{
    for(int i=0;i<r;i++)
        for(int j=0;j<c;j++)
            A[i][j]=rand()%10;
}

void serialMultiply(int **A,int **B,int **C,int m,int n,int p)
{
    clock_t start=clock();

    for(int i=0;i<m;i++)
        for(int j=0;j<p;j++)
        {
            C[i][j]=0;
            for(int k=0;k<n;k++)
                C[i][j]+=A[i][k]*B[k][j];
        }

    clock_t end=clock();
    printf("\nSerial Time = %.3f ms\n",
           ((double)(end-start)/CLOCKS_PER_SEC)*1000);
}

{

    {

        {
            printf("Enter Matrix A:\n");
            manual(A,m,n);
            printf("Enter Matrix B:\n");
            manual(B,n,p);
        }
        else
        {
            randomFill(A,m,n);
            randomFill(B,n,p);
            printf("Random matrices generated.\n");
        }

    }
    return 0;
}

void parallelMultiply(int **A,int **B,int m,int n,int p)
{
    int shmid1=shmget(IPC_PRIVATE,sizeof(int)*m*p,IPC_CREAT|0666);
    int (*C)[p]=shmat(shmid1,NULL,0);

    int shmid2=shmget(IPC_PRIVATE,sizeof(double)*m,IPC_CREAT|0666);
    double *T=shmat(shmid2,NULL,0);

    for(int i=0;i<m;i++)
    {
        if(fork()==0)
        {
            clock_t start=clock();

            for(int j=0;j<p;j++)
            {
                C[i][j]=0;
                for(int k=0;k<n;k++)
                    C[i][j]+=A[i][k]*B[k][j];
            }

            clock_t end=clock();
            T[i]=((double)(end-start)/CLOCKS_PER_SEC)*1000;
            exit(0);
        }
    }

    for(int i=0;i<m;i++) wait(NULL);

    double max=T[0];
    for(int i=1;i<m;i++)
        if(T[i]>max) max=T[i];

    printf("\nParallel Completion Time = %.3f ms\n",max);

    shmdt(C); shmdt(T);
    shmctl(shmid1,IPC_RMID,NULL);
    shmctl(shmid2,IPC_RMID,NULL);
}
